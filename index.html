crontab -e @reboot xdg-open "https://claude.ai/code/artifact/8002ebd9-e494-45af-a5f5-036e9b51a6b2" 0 7 * * * xdg-open "https://claude.ai/code/artifact/8002ebd9-e494-45af-a5f5-036e9b51a6b2"
